14: function AreaChart({ data }) {
15:   if (!data || data.length === 0) return <div className="py-16 text-center text-sm text-gray-400">No trend data</div>;
16:   const w = 640, h = 240, pad = 36;
17:   const max = Math.max(...data.map((d) => Number(d.total || 0)), 1);
18:   const stepX = (w - pad * 2) / (data.length - 1 || 1);
19:   const pts = data.map((d, i) => {
20:     const x = pad + i * stepX;
21:     const y = h - pad - (Number(d.total || 0) / max) * (h - pad * 2);
22:     return { x, y, d };
23:   });
24:   const line = pts.map((p) => `${p.x},${p.y}`).join(' ');
25:   const area = `${pad},${h - pad} ${line} ${w - pad},${h - pad}`;
26:   return (
27:     <svg viewBox={`0 0 ${w} ${h}`} className="w-full h-52">
28:       <defs>
29:         <linearGradient id="areaGrad" x1="0" y1="0" x2="0" y2="1">
30:           <stop offset="0%" stopColor="#0d9488" stopOpacity="0.35" />
31:           <stop offset="100%" stopColor="#0d9488" stopOpacity="0.02" />
32:         </linearGradient>
33:       </defs>
34:       {[0, 0.25, 0.5, 0.75, 1].map((f, i) => {
35:         const y = pad + f * (h - pad * 2);
36:         const val = max - max * f;
37:         return (
38:           <g key={i}>
39:             <line x1={pad} x2={w - pad} y1={y} y2={y} stroke="#e5e7eb" strokeWidth="1" className="dark:stroke-gray-700" />
40:             <text x={pad - 6} y={y + 4} textAnchor="end" fontSize="9" fill="#9ca3af">{formatMoney(val)}</text>
41:           </g>
42:         );
43:       })}
44:       <polygon points={area} fill="url(#areaGrad)" />
45:       <polyline points={line} fill="none" stroke="#0d9488" strokeWidth="2.5" strokeLinecap="round" strokeLinejoin="round" />
46:       {pts.map((p, i) => (
47:         <g key={i}>
48:           <circle cx={p.x} cy={p.y} r="3.5" fill="#0d9488" stroke="#fff" strokeWidth="1.5" className="dark:stroke-gray-800" />

51:       ))}
52:     </svg>
53:   );
54: }

56: function Donut({ data }) {
57:   const segs = [
58:     ['Cash', data?.cash, COLORS.cash],
59:     ['Momo', data?.momo, COLORS.momo],
60:     ['Credit', data?.credit, COLORS.credit],
61:     ['POS', data?.pos, COLORS.pos],
62:     ['E-Kashi', data?.ekashi, COLORS.ekashi]
63:   ];
64:   const total = segs.reduce((s, [, v]) => s + Number(v || 0), 0);
65:   if (!total) return <div className="py-16 text-center text-sm text-gray-400">No data</div>;
66:   const r = 46, c = 2 * Math.PI * r;
67:   const rendered = segs
68:     .filter(([, v]) => Number(v || 0) > 0)
69:     .map(([label, v, color]) => {
70:       const dash = (Number(v) / total) * c;
71:       return { label, color, dash, pct: Math.round((Number(v) / total) * 100) };
72:     });
73:   let offset = 0;
74:   const circles = rendered.map((s) => {
75:     const item = { ...s, offset };
76:     offset -= s.dash;
77:     return item;
78:   });
79:   return (
80:     <div className="flex flex-col items-center gap-4">
81:       <div className="relative w-40 h-40">
82:         <svg viewBox="0 0 120 120" className="w-40 h-40 -rotate-90">
83:           <circle cx="60" cy="60" r={r} fill="none" stroke="#e5e7eb" strokeWidth="18" className="dark:stroke-gray-700" />
84:           {circles.map((s, i) => (
85:             <circle key={i} cx="60" cy="60" r={r} fill="none" stroke={s.color} strokeWidth="18"
86:               strokeDasharray={`${s.dash} ${c - s.dash}`} strokeDashoffset={s.offset} />
87:           ))}
88:           <g transform="rotate(90 60 60)">
89:             <text x="60" y="57" textAnchor="middle" fontSize="12" fontWeight="bold" fill="#111827" className="dark:fill-gray-100">{formatMoney(total)}</text>
90:             <text x="60" y="72" textAnchor="middle" fontSize="7" fill="#6b7280">Total</text>
91:           </g>
92:         </svg>
93:       </div>
94:       <div className="w-full grid grid-cols-1 sm:grid-cols-2 gap-2">
95:         {circles.map((s, i) => (
96:           <div key={i} className="flex items-center justify-between text-xs">
97:             <span className="flex items-center gap-1.5 text-gray-600 dark:text-gray-300">
98:               <span className="w-2.5 h-2.5 rounded-full" style={{ backgroundColor: s.color }} />{s.label}
99:             </span>
100:             <span className="font-semibold text-gray-800 dark:text-gray-100">{s.pct}%</span>
101:           </div>
102:         ))}
103:       </div>
104:     </div>
105:   );
106: }

108: function Progress({ label, value, max, color }) {
109:   const solid = Number(value || 0) || 0;

111:   return (
112:     <div>

114:         <span className="text-gray-600 dark:text-gray-300 truncate pr-2">{label}</span>
115:         <span className="font-semibold text-gray-800 dark:text-gray-100">{formatMoney(solid)}</span>
116:       </div>
117:       <div className="h-2.5 bg-gray-100 dark:bg-gray-700 rounded-full overflow-hidden">
118:         <div className="h-full rounded-full transition-all" style={{ width: `${pct}%`, backgroundColor: color }} />
119:       </div>
120:     </div>
121:   );
122: }
